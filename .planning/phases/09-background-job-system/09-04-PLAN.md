    ```python
    @pytest.mark.asyncio
    async def test_disconnect_dead_within_200ms(case_dir_fixture, registry_factory, monkeypatch):
        from mcp_gateway import jobs as J
        from mcp_gateway import session_state
        from mcp_gateway.tools.case_dirs import resolve_case_dir
        import asyncio, os, time

        async with registry_factory(cancel_grace_s=0.05) as reg:
            session_state.JOB_REGISTRY = reg
            try:
                spec = J.JOB_TOOL_REGISTRY["_sleep_probe"]
                job = await reg.submit(
                    spec=spec, kwargs={"seconds": 30},
                    case_dir_resolved=str(case_dir_fixture),
                    effective_timeout_s=60.0,
                )
                # Wait for spawn to populate proc/pgid
                for _ in range(100):
                    if job.proc is not None and job.pgid is not None:
                        break
                    await asyncio.sleep(0.01)
                assert job.proc is not None and job.pgid is not None
                captured_pid = job.proc.pid

                t0 = time.monotonic()
                # Simulate SC-4: drive task externally cancelled (e.g., registry shutdown)
                job._drive_task.cancel()
                try:
                    await job._drive_task
                except asyncio.CancelledError:
                    pass

                # Assert subprocess reaped within 200 ms
                for _ in range(40):  # 40 * 5 ms = 200 ms
                    try:
                        os.kill(captured_pid, 0)
                    except ProcessLookupError:
                        elapsed = time.monotonic() - t0
                        assert elapsed < 0.2, f"reaped at {elapsed*1000:.1f} ms (> 200 ms)"
                        return
                    await asyncio.sleep(0.005)
                pytest.fail("subprocess survived 200 ms after drive-task cancellation")
            finally:
                session_state.JOB_REGISTRY = None
    ```
