<objective>
Land the seven `@mcp.tool()` handlers for dynamic mode in `mcp-gateway/src/mcp_gateway/tools/dynamic.py`, plus the env-gate wiring in `tools/__init__.py::register_all_tools`. The MCP surface is the only client-facing layer — every primitive (capability probes, argv builders, gdb driver, session registry, JOBS dispatcher) lands in Plans 01-03.

Each handler:
1. Resolves `case_dir` via `tools.case_dirs.resolve_case_dir` (existing helper).
2. Resolves `sample_sha256` via `tools.samples.resolve_sample` (returns absolute path; raises FileNotFoundError on miss — caught and converted to structured error).
3. Consults `dynamic.CAPABILITIES` slot; returns structured `{error: 'dynamic capability unavailable', missing: [...], hint: '...'}` if prerequisites missing.
4. Dispatches via `start_tool_job` (3 trace tools) or `SessionRegistry.open(kind='gdb', ...)` (gdb tools).
5. NEVER raises out of the MCP boundary — wraps everything in try/except returning structured error dicts.
6. Docstring includes the D-DYN-TOOL-02 disclaimer string (regression-tested).

Also extends `test_tool_list.py::EXPECTED_TOOLS` to be PARAMETRIZED on `MCP_GATEWAY_DYNAMIC_TOOLS` (54 when unset, 61 when "1"). Tool count assertion remains the load-bearing invariant for DYN-01.

Purpose: deliver the user-facing tool surface that Plans 01-03 made possible.
Output: tools/dynamic.py (~500 LoC), tools/__init__.py 5-line env-gate edit, tests/test_dynamic_tools.py + tests/test_dynamic_gate.py.

    async def gdb_exec(
        session_id: str,
        cmd: str,
        timeout: Optional[float] = None,
    ) -> dict:
        """Execute one gdb-MI3 command in an open session; allowlisted prefixes only."""
        registry = session_state.SESSION_REGISTRY
        if registry is None:
            return {"error": "session registry not initialized"}
        try:
            sess = registry.get(session_id)
        except KeyError:
            return {"error": "unknown session_id", "session_id": session_id}
        try:
            validate_mi_command(cmd)
        except ValueError as e:
            return {"error": "gdb command refused", "reason": str(e), "session_id": session_id}
        timeout_s = float(timeout) if timeout is not None else GDB_CMD_TIMEOUT_S
        slug = "gdb_cmd"
        t0 = time.monotonic()
        try:
            async with sess.lock:
                raw, timed_out = await sess.exec_one(cmd, timeout=timeout_s)
                sess.command_count += 1
                sess.last_used_at = time.monotonic()
        except Exception as e:
            return _err_internal(e)
        duration = time.monotonic() - t0
        # Best-effort MI parse: locate `^done|^error|^running|^connected|^exit` result-class line
        mi_result_class = "unknown"
        mi_records: list[dict] = []
        parse_error: Optional[str] = None
        try:
            for line in raw.split(b"\n"):
                s = line.decode("utf-8", errors="replace").strip()
                if s.startswith("^done"):
                    mi_result_class = "done"
                elif s.startswith("^error"):
                    mi_result_class = "error"
                elif s.startswith("^running"):
                    mi_result_class = "running"
                elif s.startswith("^connected"):
                    mi_result_class = "connected"
                elif s.startswith("^exit"):
                    mi_result_class = "exit"
        except Exception as e:
            parse_error = f"{type(e).__name__}: {e}"
        # If we timed out, invalidate the session per D-09
        session_invalidated = False
        if timed_out:
            session_invalidated = True
            try:
                await registry.close(session_id, reason="cmd_timeout")
            except Exception:
                log.exception("[tools/dynamic] failed to close invalidated session %s", session_id)
        # Append transcript line
        try:
            with open(sess.transcript_path, "ab") as f:
                f.write(f"\n--- gdb_cmd {datetime.datetime.now(datetime.timezone.utc).isoformat(timespec='seconds')} ---\n".encode())
                f.write(cmd.encode("utf-8") + b"\n")
                f.write(raw)
                f.write(b"\n")
        except OSError:
            log.exception("[tools/dynamic] transcript append failed")
        stdout_text = raw.decode("utf-8", errors="replace")
        # Truncate to head_kb (use Phase 8 truncate helper)
        from mcp_gateway.sessions import truncate_for_response
        stdout_head = truncate_for_response(stdout_text, 32)
        return {
            "exit_code": 0 if not timed_out else -9,
            "timed_out": timed_out,
            "duration_s": duration,
            "stdout_head": stdout_head,
            "stdout_truncated": len(stdout_text.encode("utf-8")) > 32 * 1024,
            "stdout_bytes_total": len(stdout_text.encode("utf-8")),
            "stderr_head": "",
            "stderr_truncated": False,
            "stderr_bytes_total": 0,
            "log_path": str(sess.transcript_path.relative_to(sess.case_dir)),
            "argv": ["gdb-session-cmd", session_id[:8], cmd[:80]],
            "slug": slug,
            "session_id": session_id,
            "session_invalidated": session_invalidated,
            "transcript_path": str(sess.transcript_path.relative_to(sess.case_dir)),
            "mi_result_class": mi_result_class,
            "mi_records": mi_records,
            "parse_error": parse_error,
        }
    gdb_exec.__doc__ = (gdb_exec.__doc__ or "") + _DYNAMIC_TOOL_DISCLAIMER
