    **test_cancel_propagates_to_killpg** (Pitfall 18):
    ```python
    @pytest.mark.asyncio
    async def test_cancel_propagates_to_killpg(session_registry, open_case, monkeypatch):
        # ... open session ...
        # Capture the r2 PID BEFORE issuing aaaa, so we can verify it dies post-cancel.
        sess_obj = session_registry.get(sid)
        pid = sess_obj.proc.pid

        task = asyncio.create_task(r2_sessions.r2_cmd(sid, "aaaa", timeout=60.0))
        await asyncio.sleep(0.5)
        task.cancel()
        try:
            await task
        except asyncio.CancelledError:
            pass

        # Pitfall 18: PID must be dead within 200ms of cancel.
        for _ in range(20):  # 20 * 10ms = 200ms
            try:
                os.kill(pid, 0)
                await asyncio.sleep(0.01)
            except ProcessLookupError:
                break
        else:
            raise AssertionError(f"r2 pid {pid} still alive 200ms after cancel")
    ```
