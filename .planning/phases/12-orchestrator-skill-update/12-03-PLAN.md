    **Add the function (after the existing `touch "$CASE_DIR/$f"` loop but BEFORE the final `echo "Initialized case directory: $CASE_DIR"` line):**

    ```bash
    # ---------------------------------------------------------------------------
    # Phase 12 / D-16: populate dynamic-cap fields in CURRENT_STATE.json.
    # Mode is detected via the MCP gateway HTTP endpoint; falls back to the
    # project-root scripts/probe_dynamic_tools.sh in scripts mode.
    # Failures degrade gracefully: dynamic_mode_enabled=false + INDEX.md note.
    # ---------------------------------------------------------------------------
    populate_dynamic_caps() {
      local case_dir="$1"
      local mode="scripts"
      local dyn_enabled="false"
      local dyn_caps_json="{}"
      local probe_note=""

      # T-12-02: bearer token from env var ONLY. Never embed literal tokens.
      local token="${MCP_GATEWAY_TOKEN:-}"
      local port="${MCP_GATEWAY_HOST_PORT:-8080}"
      local host="${MCP_GATEWAY_HOST:-127.0.0.1}"

      # Gateway-mode probe: only attempt if token is set AND curl can reach /healthz.
      # Pitfall 6: --max-time + --retry to avoid startup-window deadlock.
      if [[ -n "$token" ]] && command -v curl >/dev/null 2>&1 && command -v jq >/dev/null 2>&1; then
        if curl -sf --max-time 3 --retry 3 --retry-delay 1 --retry-connrefused \
                -H "Authorization: Bearer $token" \
                "http://${host}:${port}/healthz" >/dev/null 2>&1; then
          mode="gateway"
          local resp
          resp=$(curl -sf --max-time 5 -X POST \
                   -H "Authorization: Bearer $token" \
                   -H "Content-Type: application/json" \
                   -d '{"jsonrpc":"2.0","id":1,"method":"tools/call","params":{"name":"get_dynamic_capabilities","arguments":{}}}' \
                   "http://${host}:${port}/mcp" 2>/dev/null || echo '')
          if [[ -n "$resp" ]]; then
            # Extract the embedded JSON text payload (FastMCP serializes tool results as text/content).
            local payload
            payload=$(echo "$resp" | jq -r '.result.content[0].text // empty' 2>/dev/null || echo '')
            if [[ -n "$payload" ]]; then
              dyn_enabled=$(echo "$payload" | jq -r '.dynamic_mode_enabled // false' 2>/dev/null || echo 'false')
              dyn_caps_json=$(echo "$payload" | jq -c '{
                ptrace_scope: (.ptrace_scope // null),
                binfmt_misc: (.binfmt_misc_mounted // false),
                qemu_archs: (.qemu_architectures // []),
                netns_feasible: (.netns_feasible // false)
              }' 2>/dev/null || echo '{}')
            else
              probe_note="gateway responded but get_dynamic_capabilities payload was empty"
            fi
          else
            probe_note="gateway healthz OK but tools/call timed out"
          fi
        fi
      fi

      # Scripts-mode fallback (and the case where the gateway probe above didn't set mode=gateway).
      if [[ "$mode" == "scripts" ]]; then
        # Project root probe script (Phase 11). Path is relative to inner-agent workspace root.
        local probe_script
        probe_script="$(cd "$SKILL_DIR/../../.." 2>/dev/null && pwd)/scripts/probe_dynamic_tools.sh"
        if [[ -x "$probe_script" ]]; then
          # probe_dynamic_tools.sh emits [OK]/[WARN]/[INFO] lines + exit 0/1.
          # Grep its output for capability markers and build the JSON.
          local probe_out probe_rc
          probe_out=$("$probe_script" 2>&1 || true)
          probe_rc=$?

          # Heuristic mapping: probe_dynamic_tools.sh reports these markers
          # (see scripts/probe_dynamic_tools.sh:1-121).
          local ptrace_scope='null'
          local binfmt_misc='false'
          local qemu_archs='[]'
          local netns_feasible='false'

          # ptrace_scope: probe emits "ptrace_scope: <int>"
          ptrace_scope=$(echo "$probe_out" | grep -oP 'ptrace_scope:\s*\K-?\d+' | head -1)
          [[ -z "$ptrace_scope" ]] && ptrace_scope='null'

          # binfmt_misc: probe emits [OK] for binfmt_misc-related checks
          if echo "$probe_out" | grep -qE '\[OK\].*binfmt_misc'; then binfmt_misc='true'; fi

          # qemu_archs: probe lists each qemu-<arch>-static found
          qemu_archs=$(echo "$probe_out" | grep -oP 'qemu-\K[a-z0-9_]+(?=-static)' | jq -R . | jq -sc . 2>/dev/null || echo '[]')

          # netns_feasible: probe emits [OK] unshare -Un
          if echo "$probe_out" | grep -qE '\[OK\].*unshare'; then netns_feasible='true'; fi

          # Compose the caps JSON
          dyn_caps_json=$(jq -nc \
            --argjson ptrace_scope "$ptrace_scope" \
            --argjson binfmt_misc "$binfmt_misc" \
            --argjson qemu_archs "$qemu_archs" \
            --argjson netns_feasible "$netns_feasible" \
            '{ptrace_scope: $ptrace_scope, binfmt_misc: $binfmt_misc, qemu_archs: $qemu_archs, netns_feasible: $netns_feasible}' \
            2>/dev/null || echo '{}')

          # dynamic_mode_enabled is true only if probe exited 0 AND we have a non-empty caps
          if [[ "$probe_rc" -eq 0 && "$dyn_caps_json" != '{}' ]]; then
            dyn_enabled='true'
          else
            probe_note="probe_dynamic_tools.sh exit=$probe_rc; dynamic features unavailable"
          fi
        else
          probe_note="no MCP gateway token AND no project-root scripts/probe_dynamic_tools.sh found"
        fi
      fi

      # Splice into CURRENT_STATE.json via update_state.py (D-17).
      python3 "$SCRIPT_DIR/update_state.py" --status-dir "$case_dir" --phase init \
              --probe-dynamic --mode "$mode" \
              --dynamic-enabled "$dyn_enabled" --dynamic-caps "$dyn_caps_json" >/dev/null

      # If the probe failed, append a note to INDEX.md (D-16 INDEX.md note).
      if [[ -n "$probe_note" ]]; then
        {
          echo ""
          echo "## Dynamic-mode probe note"
          echo "- mode: $mode"
          echo "- dynamic_mode_enabled: $dyn_enabled"
          echo "- note: $probe_note"
        } >> "$case_dir/INDEX.md"
      fi
    }

    # Invoke once at case-init time. Safe to re-run via update_state.py --probe-dynamic (D-17).
    populate_dynamic_caps "$CASE_DIR"
    ```
