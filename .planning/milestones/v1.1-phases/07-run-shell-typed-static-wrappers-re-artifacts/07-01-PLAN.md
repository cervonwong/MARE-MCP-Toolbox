  # Fallback: handcraft a minimal PE header so 'file' identifies it as PE32+.
  # 'detect-it-easy' may flag it as not-fully-formed but the magic + DOS stub suffice for run_file / run_die smoke tests.

import struct
with open("hello_pe.exe", "wb") as f:
    # DOS header
    f.write(b"MZ" + b"\x00" * 58 + struct.pack("<I", 0x80))  # e_lfanew = 0x80
